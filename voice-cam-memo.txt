  <script>
    // --- 状態管理 ---
    let isRecording = false;
    let recognition = null;
    let stream = null;
    let entries = []; // { id, type: 'photo'|'text', time, image?, text? }
    let photoCount = 0;
    let lastTriggerAt = 0;
    let interimEl = null;

    const video = document.getElementById('video');
    const canvas = document.getElementById('canvas');
    const flash = document.getElementById('flash');
    const streamList = document.getElementById('streamList');
    const startStopBtn = document.getElementById('startStopBtn');
    const manualSnapBtn = document.getElementById('manualSnapBtn');
    const toggleViewBtn = document.getElementById('toggleViewBtn');
    const captureView = document.getElementById('captureView');
    const reviewView = document.getElementById('reviewView');
    const reviewList = document.getElementById('reviewList');
    const recDot = document.getElementById('recDot');
    const statusText = document.getElementById('statusText');
    const photoCountPill = document.getElementById('photoCountPill');
    const speechStatePill = document.getElementById('speechStatePill');
    const triggerWordsInput = document.getElementById('triggerWords');
    const moodPreset = document.getElementById('moodPreset');

    // --- 気分・現場プリセット切替 ---
    moodPreset.addEventListener('change', () => {
      const v = moodPreset.value;
      if (v === 'quick') {
        triggerWordsInput.value = 'し,はい,次,撮る,写真';
      } else if (v === 'standard') {
        triggerWordsInput.value = '写真,撮る,シャッター,続けて';
      } else if (v === 'strict') {
        triggerWordsInput.value = 'シャッターチャンス,写真撮影,次の工程';
      }
    });

    // --- 時刻フォーマット ---
    function nowStr() {
      return new Date().toLocaleTimeString('ja-JP', { hour12: false });
    }

    // --- カメラ初期化 ---
    async function initCamera() {
      if (stream) return;
      try {
        stream = await navigator.mediaDevices.getUserMedia({
          video: { facingMode: { ideal: 'environment' }, width: { ideal: 1280 } },
          audio: false
        });
        video.srcObject = stream;
      } catch (e) {
        alert('カメラの起動に失敗しました。HTTPS環境かアクセス許可を確認してください。\n' + e.message);
      }
    }

    // --- 写真撮影処理 ---
    function takePhoto(reason = '音声') {
      if (!video.videoWidth) return;
      const now = Date.now();
      if (now - lastTriggerAt < 1500) return; // 連続誤爆防止(1.5秒)
      lastTriggerAt = now;

      // フラッシュ演出
      flash.classList.add('show');
      setTimeout(() => flash.classList.remove('show'), 150);

      // キャプチャ
      canvas.width = video.videoWidth;
      canvas.height = video.videoHeight;
      const ctx = canvas.getContext('2d');
      ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
      const dataUrl = canvas.toDataURL('image/jpeg', 0.8);

      photoCount++;
      const t = nowStr();
      const entry = { id: now, type: 'photo', num: photoCount, time: t, image: dataUrl };
      entries.push(entry);
      saveLocal();

      photoCountPill.textContent = `📷 ${photoCount}枚`;

      // ストリーム画面にカメラマーカー追加
      const div = document.createElement('div');
      div.className = 'log-item photo-marker';
      div.innerHTML = `<span>📷 写真 #${photoCount} を撮影（${reason}）</span><span class="time">${t}</span>`;
      insertLogElement(div);
    }

    // --- ストリームログ挿入 ---
    function insertLogElement(el) {
      if (interimEl && interimEl.parentNode === streamList) {
        streamList.insertBefore(el, interimEl);
      } else {
        streamList.appendChild(el);
      }
      streamList.scrollTop = streamList.scrollHeight;
    }

    function appendTextLog(text) {
      const clean = text.trim();
      if (!clean) return;
      const t = nowStr();
      entries.push({ id: Date.now() + Math.random(), type: 'text', time: t, text: clean });
      saveLocal();

      const div = document.createElement('div');
      div.className = 'log-item';
      div.innerHTML = `<span class="time">${t}</span>${escapeHtml(clean)}`;
      insertLogElement(div);
    }

    // --- 音声認識 (iOS Safari対応・自動再起動付き) ---
    function setupSpeech() {
      const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
      if (!SpeechRecognition) {
        alert('このブラウザは音声認識（Web Speech API）に対応していません。iPhoneのSafariでお試しください。');
        return null;
      }
      const rec = new SpeechRecognition();
      rec.lang = 'ja-JP';
      rec.continuous = true;
      rec.interimResults = true;

      rec.onstart = () => {
        speechStatePill.textContent = '🎤 音声認識中';
      };

      rec.onresult = (event) => {
        let interimTranscript = '';
        for (let i = event.resultIndex; i < event.results.length; i++) {
          const res = event.results[i];
          const transcript = res[0].transcript;

          if (res.isFinal) {
            if (interimEl) {
              interimEl.remove();
              interimEl = null;
            }
            handleRecognizedPhrase(transcript);
          } else {
            interimTranscript += transcript;
            // 仮認識中でもトリガーワードが含まれたら即反応させる場合
            checkTriggerQuick(transcript);
          }
        }

        if (interimTranscript) {
          if (!interimEl) {
            interimEl = document.createElement('div');
            interimEl.className = 'log-item interim';
            streamList.appendChild(interimEl);
          }
          interimEl.textContent = '… ' + interimTranscript;
          streamList.scrollTop = streamList.scrollHeight;
        }
      };

      rec.onerror = (e) => {
        if (e.error !== 'no-speech' && e.error !== 'aborted') {
          speechStatePill.textContent = '⚠️ 音声エラー: ' + e.error;
        }
      };

      // iOS Safariは一定時間や無音で自動終了するため、収録中なら即再開
      rec.onend = () => {
        if (isRecording) {
          try {
            rec.start();
          } catch (_) {}
        } else {
          speechStatePill.textContent = '🎤 マイク停止中';
        }
      };

      return rec;
    }

    function getTriggerList() {
      return triggerWordsInput.value
        .split(/[,、]/)
        .map(w => w.trim())
        .filter(Boolean);
    }

    function checkTriggerQuick(text) {
      const triggers = getTriggerList();
      for (const word of triggers) {
        if (word.length >= 2 && text.endsWith(word)) {
          takePhoto(`音声:「${word}」`);
          break;
        }
      }
    }

    function handleRecognizedPhrase(text) {
      const triggers = getTriggerList();
      let triggered = false;
      let cleanedText = text;

      for (const word of triggers) {
        if (cleanedText.includes(word)) {
          triggered = true;
          // 単独トリガー発話の場合はテキストから除去、文中の場合は残す
          if (cleanedText.trim() === word) {
            cleanedText = '';
          }
        }
      }

      if (triggered) {
        takePhoto('音声トリガー');
      }
      if (cleanedText.trim()) {
        appendTextLog(cleanedText);
      }
    }

    // --- 収録開始 / 停止 ---
    startStopBtn.addEventListener('click', async () => {
      if (!isRecording) {
        await initCamera();
        if (!recognition) recognition = setupSpeech();
        isRecording = true;
        try { recognition && recognition.start(); } catch (_) {}

        recDot.classList.add('active');
        statusText.textContent = '収録中';
        startStopBtn.textContent = '⏹ 一時停止';
        startStopBtn.className = 'btn-danger';
      } else {
        isRecording = false;
        try { recognition && recognition.stop(); } catch (_) {}

        recDot.classList.remove('active');
        statusText.textContent = '一時停止';
        startStopBtn.textContent = '▶ 収録再開';
        startStopBtn.className = 'btn-green';
      }
    });

    manualSnapBtn.addEventListener('click', async () => {
      await initCamera();
      takePhoto('手動');
    });

    // --- 画面切替（収録 ⇔ 確認・マニュアル作成） ---
    toggleViewBtn.addEventListener('click', () => {
      const isCapture = captureView.classList.contains('active');
      if (isCapture) {
        captureView.classList.remove('active');
        reviewView.classList.add('active');
        toggleViewBtn.textContent = '← カメラに戻る';
        renderReview();
      } else {
        reviewView.classList.remove('active');
        captureView.classList.add('active');
        toggleViewBtn.textContent = '確認・作成へ';
      }
    });

    // --- 時系列ブロック化＆確認画面描画 ---
    function buildManualBlocks() {
      const blocks = [];
      let currentBlock = { title: '導入メモ（撮影前）', time: '', image: null, texts: [], entryIds: [] };

      for (const item of entries) {
        if (item.type === 'photo') {
          if (currentBlock.image || currentBlock.texts.length > 0) {
            blocks.push(currentBlock);
          }
          currentBlock = {
            title: `ステップ ${item.num}`,
            time: item.time,
            image: item.image,
            texts: [],
            entryIds: [item.id]
          };
        } else if (item.type === 'text') {
          currentBlock.texts.push(item.text);
          currentBlock.entryIds.push(item.id);
        }
      }
      if (currentBlock.image || currentBlock.texts.length > 0) {
        blocks.push(currentBlock);
      }
      return blocks;
    }

    function renderReview() {
      const blocks = buildManualBlocks();
      reviewList.innerHTML = '';

      if (blocks.length === 0) {
        reviewList.innerHTML = '<div class="empty-msg">まだ記録データがありません。カメラ画面で撮影・音声入力を行ってください。</div>';
        return;
      }

      blocks.forEach((b, idx) => {
        const card = document.createElement('div');
        card.className = 'card';
        card.innerHTML = `
          <div class="card-header">
            <span>${escapeHtml(b.title)} ${b.time ? `(${b.time})` : ''}</span>
            <span>#${idx + 1}</span>
          </div>
          ${b.image ? `<img src="${b.image}" alt="${escapeHtml(b.title)}" />` : ''}
          <textarea data-block="${idx}" placeholder="音声コメントがここに入ります（手修正も可能）">${escapeHtml(b.texts.join('\n'))}</textarea>
        `;
        reviewList.appendChild(card);
      });
    }

    // --- Markdownコピー ---
    document.getElementById('copyMdBtn').addEventListener('click', async () => {
      const blocks = buildManualBlocks();
      const textareas = reviewList.querySelectorAll('textarea');
      let md = `# 作業記録・マニュアル (${new Date().toLocaleDateString('ja-JP')})\n\n`;

      blocks.forEach((b, idx) => {
        const editedText = textareas[idx] ? textareas[idx].value.trim() : b.texts.join('\n');
        md += `## ${b.title} ${b.time ? `[${b.time}]` : ''}\n`;
        if (b.image) md += `[写真データあり]\n`;
        md += `${editedText || '（コメントなし）'}\n\n`;
      });

      await navigator.clipboard.writeText(md);
      alert('時系列マニュアルをクリップボードにコピーしました！');
    });

    // --- JSON保存（後の構造化データ用） ---
    document.getElementById('downloadJsonBtn').addEventListener('click', () => {
      const blocks = buildManualBlocks();
      const textareas = reviewList.querySelectorAll('textarea');
      const payload = blocks.map((b, idx) => ({
        step: idx + 1,
        title: b.title,
        timestamp: b.time,
        comment: textareas[idx] ? textareas[idx].value.trim() : b.texts.join('\n'),
        hasImage: Boolean(b.image),
        imageBase64: b.image || null
      }));

      const blob = new Blob([JSON.stringify(payload, null, 2)], { type: 'application/json' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `field-log-${Date.now()}.json`;
      a.click();
      URL.revokeObjectURL(url);
    });

    // --- 全消去 ---
    document.getElementById('clearAllBtn').addEventListener('click', () => {
      if (!confirm('記録した写真と音声ログをすべて消去しますか？')) return;
      entries = [];
      photoCount = 0;
      localStorage.removeItem('voice_cam_proto_v1');
      photoCountPill.textContent = '📷 0枚';
      streamList.innerHTML = '<div class="log-item">ログをリセットしました。</div>';
      renderReview();
    });

    // --- ローカル一時保存 ---
    function saveLocal() {
      try {
        localStorage.setItem('voice_cam_proto_v1', JSON.stringify({ entries, photoCount }));
      } catch (e) {
        console.warn('容量上限のため一部保存スキップ:', e);
      }
    }

    function loadLocal() {
      try {
        const raw = localStorage.getItem('voice_cam_proto_v1');
        if (!raw) return;
        const parsed = JSON.parse(raw);
        entries = parsed.entries || [];
        photoCount = parsed.photoCount || 0;
        photoCountPill.textContent = `📷 ${photoCount}枚`;

        if (entries.length > 0) {
          streamList.innerHTML = '';
          entries.forEach(item => {
            const div = document.createElement('div');
            if (item.type === 'photo') {
              div.className = 'log-item photo-marker';
              div.innerHTML = `<span>📷 写真 #${item.num} を撮影</span><span class="time">${item.time}</span>`;
            } else {
              div.className = 'log-item';
              div.innerHTML = `<span class="time">${item.time}</span>${escapeHtml(item.text)}`;
            }
            streamList.appendChild(div);
          });
        }
      } catch (_) {}
    }

    function escapeHtml(str) {
      return str.replace(/[&<>"']/g, m => ({ '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;' }[m]));
    }

    loadLocal();
  </script>
